const SHEET_NAME = "records"; const CONFIG_SHEET = "config"; function doGet(e) { const p = e.parameter; const callback = p.callback; // JSONP callback const action = p.action || "getAll"; let result; try { if (action === "getAll") result = getAll(); else if (action === "getConfig") result = getConfig(); else if (action === "saveRecord") result = saveRecord(JSON.parse(p.record)); else if (action === "deleteRecord") result = deleteRecord(p.id); else if (action === "settleRecord") result = settleRecord(JSON.parse(p.ids), p.settled === "true"); else if (action === "saveConfig") result = saveConfig( p.items ? JSON.parse(p.items) : null, p.methods ? JSON.parse(p.methods) : null ); else result = { error: "unknown action" }; } catch(err) { result = { error: err.message }; } const json = JSON.stringify(result); // 如果有 callback 參數就用 JSONP，否則回傳一般 JSON if (callback) { return ContentService .createTextOutput(`${callback}(${json})`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headers) sheet.appendRow(headers); } return sheet; } function getAll() { const sheet = getOrCreateSheet(SHEET_NAME, ["id","date","item","amount","method","note","settled"]); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return { records: [] }; const headers = data[0]; const records = data.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i]); obj.amount = Number(obj.amount); obj.settled = obj.settled === true || obj.settled === "TRUE"; return obj; }); return { records }; } function saveRecord(r) { const sheet = getOrCreateSheet(SHEET_NAME, ["id","date","item","amount","method","note","settled"]); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(r.id)) { sheet.getRange(i+1,1,1,7).setValues([[r.id,r.date,r.item,r.amount,r.method,r.note||"",r.settled||false]]); return { ok: true, action: "updated" }; } } sheet.appendRow([r.id,r.date,r.item,r.amount,r.method,r.note||"",r.settled||false]); return { ok: true, action: "created" }; } function deleteRecord(id) { const sheet = getOrCreateSheet(SHEET_NAME, ["id","date","item","amount","method","note","settled"]); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id)) { sheet.deleteRow(i+1); return { ok: true }; } } return { error: "not found" }; } function settleRecord(ids, settled) { const sheet = getOrCreateSheet(SHEET_NAME, ["id","date","item","amount","method","note","settled"]); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (ids.map(String).includes(String(data[i][0]))) { sheet.getRange(i+1,7).setValue(settled); } } return { ok: true }; } function getConfig() { const sheet = getOrCreateSheet(CONFIG_SHEET, ["key","value"]); const data = sheet.getDataRange().getValues(); const config = {}; data.slice(1).forEach(row => { config[row[0]] = row[1]; }); return { items: config.items ? JSON.parse(config.items) : null, methods: config.methods ? JSON.parse(config.methods) : null, }; } function saveConfig(items, methods) { const sheet = getOrCreateSheet(CONFIG_SHEET, ["key","value"]); const data = sheet.getDataRange().getValues(); const keyRows = {}; data.slice(1).forEach((row, i) => { keyRows[row[0]] = i+2; }); if (items !== null) { if (keyRows["items"]) sheet.getRange(keyRows["items"],2).setValue(JSON.stringify(items)); else sheet.appendRow(["items", JSON.stringify(items)]); } if (methods !== null) { if (keyRows["methods"]) sheet.getRange(keyRows["methods"],2).setValue(JSON.stringify(methods)); else sheet.appendRow(["methods", JSON.stringify(methods)]); } return { ok: true }; }
